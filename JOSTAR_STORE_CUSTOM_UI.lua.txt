local filtergc = filtergc
if not filtergc then
    filtergc = function(ftype, filters, returnSingle)
        local results = {}
        for _, obj in pairs(getgc(true)) do
            if type(obj) == "function" then
                local ok, info = pcall(debug.getinfo, obj)
                if ok and info then
                    local match = true
                    if filters.StartLine and info.linedefined ~= filters.StartLine then match = false end
                    if filters.Source and info.source ~= filters.Source then match = false end
                    if match then
                        if returnSingle then return obj end; table.insert(results, obj)
                    end
                end
            end
        end
        if returnSingle then return nil end; return results
    end
end

if not bit32 then bit32 = {} end
if not bit32.bxor then
    bit32.bxor = function(a, b)
        local result, bit = 0, 1
        while a > 0 or b > 0 do
            if (a % 2) ~= (b % 2) then result = result + bit end
            a, b, bit = math.floor(a / 2), math.floor(b / 2), bit * 2
        end
        return result
    end
end

local function GetCommaValue(n)
    local s = tostring(math.floor(n))
    while true do
        local result, count = s:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
        s = result
        if count == 0 then break end
    end
    return s
end

local function FormatRuntime(seconds)
    return string.format("%02d:%02d:%02d",
        math.floor(seconds / 3600),
        math.floor((seconds % 3600) / 60),
        seconds % 60)
end

local function ParseMoneyText(text)
    if not text then return nil end
    text = tostring(text)
    if not text:find("%$") then return nil end
    local cleaned = text:upper():gsub("[^%d%.KMB]", "")
    if cleaned == "" then return nil end
    local numStr = cleaned:match("^[%d%.]+")
    if not numStr then return nil end
    local num = tonumber(numStr) or 0
    local suffix = cleaned:match("[KMB]")
    if suffix == "K" then return math.floor(num * 1e3)
    elseif suffix == "M" then return math.floor(num * 1e6)
    elseif suffix == "B" then return math.floor(num * 1e9)
    end
    return math.floor(num)
end

local function ScanForMoneyLabel()
    local preferred, fallback = {}, {}
    local searchRoots = { PlayerGui }
    pcall(function()
        local cg = game:GetService("CoreGui")
        if cg then table.insert(searchRoots, cg) end
    end)
    pcall(function()
        if typeof(gethui) == "function" then
            local hidden = gethui()
            if hidden then table.insert(searchRoots, hidden) end
        end
    end)

    for _, root in ipairs(searchRoots) do
        local ok, list = pcall(function() return root:GetDescendants() end)
        if ok and list then
            for _, d in ipairs(list) do
                if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.Visible then
                    local val = ParseMoneyText(d.Text)
                    if val and not ChainContains(d, MONEY_EXCLUDE_KEYWORDS) then
                        if ChainContains(d, MONEY_KEYWORDS) then
                            table.insert(preferred, { label = d, value = val })
                        else
                            table.insert(fallback, { label = d, value = val })
                        end
                    end
                end
            end
        end
    end

    if #preferred > 0 then
        table.sort(preferred, function(a, b) return a.value > b.value end)
        return preferred[1].label, preferred[1].value
    end
    if #fallback > 0 then
        table.sort(fallback, function(a, b) return a.value > b.value end)
        return fallback[1].label, fallback[1].value
    end
    return nil, nil
end

local function AttachMoneyListener()
    local lbl, val = ScanForMoneyLabel()
    if not lbl then return end
    if _moneyCache.label == lbl and _moneyCache.label.Parent and _moneyCache.listener then
        _moneyCache.value = val; return
    end
    if _moneyCache.listener then
        pcall(function() _moneyCache.listener:Disconnect() end)
        _moneyCache.listener = nil
    end
    _moneyCache.label = lbl
    _moneyCache.value = val
    _moneyCache.listener = lbl:GetPropertyChangedSignal("Text"):Connect(function()
        local newVal = ParseMoneyText(lbl.Text)
        if not newVal then return end
        local oldVal = _moneyCache.value
        if newVal > oldVal then
            Configuration.Statistics["Cash Made"] = Configuration.Statistics["Cash Made"] + (newVal - oldVal)
        end
        _moneyCache.value = newVal
    end)
end

            local STEP_WAIT = 0.028
            local MAX_ITER  = 6000
            local iter = 0
            while not getgenv().STOP_ALL and iter < MAX_ITER do
                iter = iter + 1
                local curPos = veh:GetPivot().Position
                local dX = targetPos.X - curPos.X
                local dZ = targetPos.Z - curPos.Z
                local distH = math.sqrt(dX * dX + dZ * dZ)
                if distH <= STEP_SIZE then
                    moveVehTo(Vector3.new(targetPos.X, underY, targetPos.Z), 0.08)
                    break
                end
                local dirX = dX / distH
                local dirZ = dZ / distH
                moveVehTo(Vector3.new(curPos.X + dirX * STEP_SIZE, underY, curPos.Z + dirZ * STEP_SIZE), STEP_WAIT)
            end
            local ASCEND_STEP = 2
            local ASCEND_WAIT = 0.06
            local curPos2 = veh:GetPivot().Position
            local curY2 = curPos2.Y
            while not getgenv().STOP_ALL and curY2 < targetPos.Y do
                curY2 = math.min(targetPos.Y, curY2 + ASCEND_STEP)
                moveVehTo(Vector3.new(targetPos.X, curY2, targetPos.Z), ASCEND_WAIT)
            end
        else
            moveVehTo(targetPos)
        end

        setVehCollide(true)
        Workspace.Gravity = originalGravity
        _BLINK_NOCLIP_ALL = false
        stopGhostMode()
        fixCamera()
        task.wait(0.35)
        task.wait(0.1)
        pcall(function()
            if getgenv().LockMotorPosition then getgenv().LockMotorPosition() end
        end)
        return true
    end

    _BLINK_NOCLIP_ALL = true
    startGhostMode()
    if humanoid then humanoid.PlatformStand = true end
    Workspace.Gravity = 0
    if isUnderground then
        local underY = -4
        discreteStepTP(Vector3.new(hrp.Position.X, underY, hrp.Position.Z), 1.2, 0.06)
        discreteStepTP(Vector3.new(targetPos.X, underY, targetPos.Z), 2.5, 0.05)
        discreteStepTP(Vector3.new(targetPos.X, targetPos.Y, targetPos.Z), 1.2, 0.06)
    else
        discreteStepTP(targetPos)
    end
    Workspace.Gravity = originalGravity
    if humanoid then humanoid.PlatformStand = false end
    _BLINK_NOCLIP_ALL = false
    stopGhostMode()
    fixCamera()
    return true
end

local function LockMotorPosition()
    if _MotorLockConn then return end
    local bike = _MotorState.bike
    if not bike or not bike.Parent then return end
    _MotorLockPartsOriginal = {}
    for _, part in ipairs(bike:GetDescendants()) do
        if part:IsA("BasePart") then
            _MotorLockPartsOriginal[part] = { Anchored = part.Anchored, CanCollide = part.CanCollide }
            pcall(function()
                part.Anchored = true
                part.AssemblyLinearVelocity  = Vector3.zero
                part.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end
    _MotorLockConn = RunService.Heartbeat:Connect(function()
        local b = _MotorState.bike
        if not b or not b.Parent then UnlockMotorPosition(); return end
        if _MotorLockedRotation then
            pcall(function()
                local curPos = b:GetPivot().Position
                b:PivotTo(CFrame.new(curPos) * _MotorLockedRotation)
            end)
        end
        pcall(function()
            local bikePivot = b:GetPivot()
            local checkPos  = bikePivot.Position
            local params = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances = { b, LocalPlayer.Character }
            local downHit = workspace:Raycast(checkPos, Vector3.new(0, -3, 0), params)
            if downHit and downHit.Instance then
                local part = downHit.Instance
                local nm = (part.Name .. " " .. (part.Parent and part.Parent.Name or "")):lower()
                local isObstacle = nm:find("block") or nm:find("crate") or nm:find("box")
                                   or nm:find("brick") or nm:find("rock") or nm:find("barrel")
                if isObstacle then
                    local liftY = checkPos.Y + 1.5
                    b:PivotTo(CFrame.new(Vector3.new(checkPos.X, liftY, checkPos.Z)) * (_MotorLockedRotation or bikePivot.Rotation))
                end
            end
        end)
        for _, part in ipairs(b:GetDescendants()) do
            if part:IsA("BasePart") then
                if not part.Anchored then pcall(function() part.Anchored = true end) end
                pcall(function()
                    part.AssemblyLinearVelocity  = Vector3.zero
                    part.AssemblyAngularVelocity = Vector3.zero
                end)
            end
        end
        local char = LocalPlayer.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            pcall(function()
                hrp.AssemblyLinearVelocity  = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end)
end

local function UnlockMotorPosition()
    if _MotorLockConn then _MotorLockConn:Disconnect(); _MotorLockConn = nil end
    for part, state in pairs(_MotorLockPartsOriginal) do
        if part and part.Parent then
            pcall(function()
                part.Anchored   = state.Anchored
                part.CanCollide = state.CanCollide
            end)
        end
    end
    _MotorLockPartsOriginal = {}
end

    if not ok or not pv then return false end
    _MotorLockedRotation = pv.Rotation
    pcall(function() bike:PivotTo(CFrame.new(pv.Position) * _MotorLockedRotation) end)
    return true
end

local function FindDriveSeat(bike)
    if not bike then return nil end
    for _, name in ipairs(DRIVER_SEAT_NAMES) do
        local s = bike:FindFirstChild(name, true)
        if s and (s:IsA("VehicleSeat") or s:IsA("Seat")) then return s end
    end
    local seats = {}
    for _, d in ipairs(bike:GetDescendants()) do
        if d:IsA("VehicleSeat") or d:IsA("Seat") then table.insert(seats, d) end
    end
    if #seats == 0 then return nil end
    if #seats == 1 then return seats[1] end
    local filtered = {}
    for _, s in ipairs(seats) do
        local n = s.Name:lower()
        if not (n:find("passenger") or n:find("back") or n:find("rear") or n:find("penumpang") or n:find("belakang")) then
            table.insert(filtered, s)
        end
    end
    if #filtered == 1 then return filtered[1] end
    if #filtered > 1 then table.sort(filtered, function(a,b) return a.Position.Z < b.Position.Z end); return filtered[1] end
    return nil
end

local function IsDriverSeat(seat)
    if not seat then return false end
    local n = seat.Name:lower()
    return not (n:find("passenger") or n:find("back") or n:find("rear") or n:find("penumpang") or n:find("belakang"))
end

local function IsMyBike(model)
    if not model or not model:IsA("Model") then return false end
    local myName, myUserId = LocalPlayer.Name, tostring(LocalPlayer.UserId)
    for _, attr in ipairs({"Owner","OwnerId","UserId","Player","Creator","VehicleOwner","owner","ownerId","userId","player","creator","vehicleOwner"}) do
        local ok, val = pcall(function() return model:GetAttribute(attr) end)
        if ok and val ~= nil then
            if val == LocalPlayer or val == myUserId or val == myName or tostring(val) == myUserId or tostring(val) == myName then return true end
        end
    end
    for _, child in ipairs(model:GetDescendants()) do
        if child:IsA("ObjectValue") then
            local n = child.Name:lower()
            if n:find("owner") or n:find("creator") or n:find("player") or n:find("user") then
                if child.Value == LocalPlayer then return true end
            end
        elseif child:IsA("StringValue") then
            local n = child.Name:lower()
            if n:find("owner") or n:find("user") or n:find("player") then
                if child.Value == myName or child.Value == myUserId then return true end
            end
        end
    end
    local parent, depth = model.Parent, 0
    while parent and parent ~= Workspace and depth < 5 do
        local pn = parent.Name
        if pn == myName or pn == myUserId or pn:find(myName, 1, true) or pn:find(myUserId, 1, true) then return true end
        parent = parent.Parent; depth = depth + 1
    end
    local mn = model.Name
    if mn:find(myName, 1, true) or mn:find(myUserId, 1, true) then return true end
    return false
end

local function IsOthersBike(model)
    for _, child in ipairs(model:GetDescendants()) do
        if child:IsA("ObjectValue") then
            local n = child.Name:lower()
            if n:find("owner") or n:find("player") or n:find("creator") then
                if child.Value and child.Value:IsA("Player") and child.Value ~= LocalPlayer then return true end
            end
        end
    end
    return false
end

local function FindBike()
    local char = LocalPlayer.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    local myPos = hrp and hrp.Position or Vector3.zero
    local candidates = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") then
            local seat = FindDriveSeat(obj)
            if seat and IsDriverSeat(seat) and not IsOthersBike(obj) then
                local isMine = IsMyBike(obj)
                if not REQUIRE_OWN_BIKE or isMine then
                    table.insert(candidates, {model=obj, seat=seat, ownerPrio=isMine and 0 or 1})
                end
            end
        end
    end
    if #candidates == 0 then return nil end
    table.sort(candidates, function(a,b)
        if a.ownerPrio ~= b.ownerPrio then return a.ownerPrio < b.ownerPrio end
        local da = (a.model:GetPivot().Position - myPos).Magnitude
        local db = (b.model:GetPivot().Position - myPos).Magnitude
        return da < db
    end)
    return candidates[1].model
end

    if REQUIRE_OWN_BIKE and not IsMyBike(bike) then return false end
    local DriveSeat = FindDriveSeat(bike)
    if not DriveSeat or not IsDriverSeat(DriveSeat) then return false end

    unequipTools()
    task.wait(0.3)
    if getgenv().STOP_ALL then return false end

    local TP_VERIFY_DIST = 15
    local tpOk = false
    local tpAttempt = 0
    while not getgenv().STOP_ALL do
        tpAttempt = tpAttempt + 1
        if not DriveSeat.Parent then task.wait(0.3); continue end
        local seatPos = DriveSeat.Position
        local seatCF  = DriveSeat.CFrame
        local sidePos = seatPos + seatCF.RightVector * SIDE_OFFSET + Vector3.new(0, 2.5, 0)
        local ok = instantRespawnTP(sidePos)
        if ok then
            task.wait(0.25)
            local _, _, hrpCheck = GetLiveCharacter(2)
            if hrpCheck then
                local dist = (hrpCheck.Position - seatPos).Magnitude
                if dist <= TP_VERIFY_DIST then tpOk = true; break end
                local snapCF = CFrame.new(sidePos)
                for _ = 1, 5 do
                    if getgenv().STOP_ALL then break end
                    hrpCheck.CFrame = snapCF
                    hrpCheck.AssemblyLinearVelocity  = Vector3.zero
                    hrpCheck.AssemblyAngularVelocity = Vector3.zero
                    task.wait(0.08)
                end
                local dist2 = (hrpCheck.Position - seatPos).Magnitude
                if dist2 <= TP_VERIFY_DIST then tpOk = true; break end
            end
        end
        pcall(function()
            local c = LocalPlayer.Character
            local h = c and c:FindFirstChildOfClass("Humanoid")
            local r = c and c:FindFirstChild("HumanoidRootPart")
            if h then h.PlatformStand = false end
            if r then
                r.Anchored = false
                r.AssemblyLinearVelocity  = Vector3.zero
                r.AssemblyAngularVelocity = Vector3.zero
            end
        end)
        pcall(fixCamera)
        task.wait(0.4)
    end

    if not tpOk then return false end
    if getgenv().STOP_ALL then return false end

    task.wait(0.3)
    if getgenv().STOP_ALL then return false end

    local char, hum, hrp = GetLiveCharacter(2)
    if not char or not hum or not hrp then return false end

    if (hrp.Position - DriveSeat.Position).Magnitude > TP_VERIFY_DIST then

    char, hum, hrp = GetLiveCharacter(1)

        local seatCF2 = DriveSeat.CFrame

        for _ = 1, 4 do
            hrp.CFrame = snapCF
            hrp.AssemblyLinearVelocity = Vector3.zero
            task.wait(0.03)
        end
    end

    local Prompt
    for _, d in ipairs(DriveSeat:GetDescendants()) do
        if d:IsA("ProximityPrompt") then Prompt = d; break end
    end
    if Prompt then
        InteractPromptNormal(Prompt, HOLD_DURATION)
    else
        pcall(function() DriveSeat:Sit(hum) end)
    end

    local deadline = os.clock() + (HOLD_DURATION + 3)
    while os.clock() < deadline do
        if getgenv().STOP_ALL then return false end
        local c = LocalPlayer.Character
        local h = c and c:FindFirstChildOfClass("Humanoid")
        if h and h.SeatPart == DriveSeat then
            _MotorState.bike = bike
            _MotorState.seat = DriveSeat
            if not _MotorLockedRotation then
                local ok, pv = pcall(function() return bike:GetPivot() end)
                if ok and pv then _MotorLockedRotation = pv.Rotation end
            end
            LockMotorPosition()
            return true
        end
        task.wait(0.05)
    end

    local seatPos2   = DriveSeat.Position
    local seatCF2    = DriveSeat.CFrame
    local respawnPos = seatPos2 + seatCF2.UpVector * 1.0
    instantRespawnTP(respawnPos)
    task.wait(0.25)
    fixCamera()

    local c2 = LocalPlayer.Character
    local h2 = c2 and c2:FindFirstChildOfClass("Humanoid")
    local r2 = c2 and c2:FindFirstChild("HumanoidRootPart")
    if h2 and r2 then pcall(function() DriveSeat:Sit(h2) end) end

    local deadline2 = os.clock() + 3
    while os.clock() < deadline2 do
        if getgenv().STOP_ALL then return false end
        local c = LocalPlayer.Character
        local h = c and c:FindFirstChildOfClass("Humanoid")

            _MotorState.bike = bike
            _MotorState.seat = DriveSeat
            if not _MotorLockedRotation then
                local ok, pv = pcall(function() return bike:GetPivot() end)
                if ok and pv then _MotorLockedRotation = pv.Rotation end
            end
            LockMotorPosition()

        end
        task.wait(0.1)
    end
    return false

        if ok and pv then _MotorLockedRotation = pv.Rotation end

-- FIRE PROMPT
local MAX_PROMPT_DIST = 20
local _touchedPrompts = {}

local function RegisterPrompt(prompt)
    if not prompt or not prompt.Parent then return nil end
    if not _touchedPrompts[prompt] then
        _touchedPrompts[prompt] = {
            HoldDuration          = prompt.HoldDuration,
            RequiresLineOfSight   = prompt.RequiresLineOfSight,
            MaxActivationDistance = prompt.MaxActivationDistance,
        }
    end
    return _touchedPrompts[prompt]
end

local function RestoreAllPrompts()
    for prompt, orig in pairs(_touchedPrompts) do
        if prompt and prompt.Parent then
            pcall(function()
                prompt.HoldDuration          = orig.HoldDuration
                prompt.RequiresLineOfSight   = orig.RequiresLineOfSight
                prompt.MaxActivationDistance = orig.MaxActivationDistance
            end)
        end
    end
    _touchedPrompts = {}
end

local function getCharacterPosition()
    local _, _, hrp = GetLiveCharacter(1)
    return hrp and hrp.Position or nil
end

local function firePromptOnce(prompt)
    if not prompt or not prompt.Parent then return false end
    local orig = RegisterPrompt(prompt)
    pcall(function()
        prompt.MaxActivationDistance = 9999
        prompt.RequiresLineOfSight = false
        prompt.HoldDuration = 0
        fireproximityprompt(prompt)
    end)
    task.wait(0.04)
    if orig then
        pcall(function()
            prompt.MaxActivationDistance = orig.MaxActivationDistance
            prompt.RequiresLineOfSight   = orig.RequiresLineOfSight
            prompt.HoldDuration          = orig.HoldDuration
        end)
    end
    return true
end

local function fireAllNearOnce(pos, radius)
    radius = radius or 18
    local charPos = getCharacterPosition()
    if not charPos then return false end
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local distToPos = (obj.Position - pos).Magnitude
            local distToChar = (obj.Position - charPos).Magnitude
            if distToPos <= radius and distToChar <= MAX_PROMPT_DIST then
                local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
                if prompt then firePromptOnce(prompt); return true end
            end
        end
    end
    return false
end

local function FindAvailableATMs()
    local map = Workspace:FindFirstChild("Map")
    if not map then return {} end
    local atms = map:FindFirstChild("ATMS")
    if not atms then return {} end
    local result = {}
    for _, ATM in next, atms:GetChildren() do
        local screen = ATM:FindFirstChild("ATMScreen")
        if screen and screen:IsA("BasePart") and screen.Transparency == 0 then
            table.insert(result, ATM)
        end
    end
    return result
end

local function FindNearestAvailableATM()
    local atms = FindAvailableATMs()
    if #atms == 0 then return nil end
    local charPos = getCharacterPosition()
    if not charPos then return atms[1] end
    local nearestATM, nearestDist = nil, math.huge
    for _, ATM in ipairs(atms) do
        local pos = ATM:GetPivot().Position
        local screen = ATM:FindFirstChild("ATMScreen")
        if screen and screen:IsA("BasePart") then pos = screen.Position end
        local dist = (pos - charPos).Magnitude
        if dist < nearestDist then nearestDist = dist; nearestATM = ATM end
    end
    return nearestATM
end

local function matchPromptText(actionText, keyword)
    if not keyword then return true end
    local text = string.lower(actionText or "")
    keyword = string.lower(keyword)
    if keyword == "lock" and string.find(text, "unlock") then return false end
    return string.find(text, keyword) ~= nil
end

local function firePromptAt(pos, maxDist, keyword)
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and obj.Enabled then
            local pPos = obj.Parent and (obj.Parent:IsA("BasePart") and obj.Parent.Position or (obj.Parent:IsA("Attachment") and obj.Parent.WorldPosition))
            if pPos and (pos - pPos).Magnitude <= maxDist then
                if matchPromptText(obj.ActionText, keyword) then
                    RegisterPrompt(obj)
                    obj.RequiresLineOfSight = false
                    obj.HoldDuration = 0
                    if fireproximityprompt then
                        fireproximityprompt(obj)
                    else
                        obj:InputHoldBegin(); task.wait(0.05); obj:InputHoldEnd()
                    end
                    return true
                end
            end
        end
    end
    return false
end

local function checkPromptExistsAt(pos, maxDist, keyword)
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and obj.Enabled then
            local pPos = obj.Parent and (obj.Parent:IsA("BasePart") and obj.Parent.Position or (obj.Parent:IsA("Attachment") and obj.Parent.WorldPosition))
            if pPos and (pos - pPos).Magnitude <= maxDist then
                if matchPromptText(obj.ActionText, keyword) then return true end
            end
        end
    end
    return false
end

local function secureDoor(doorPos)
    local maxAttempts = 10
    for i = 1, maxAttempts do
        if getgenv().STOP_ALL then return false end
        if checkPromptExistsAt(doorPos, 8, "unlock") then return true end
        if checkPromptExistsAt(doorPos, 8, "lock") then
            firePromptAt(doorPos, 8, "lock")
            task.wait(0.5)
            if checkPromptExistsAt(doorPos, 8, "unlock") then
                return true
            else
                firePromptAt(doorPos, 8, "open")
                task.wait(0.7)
            end
        else
            firePromptAt(doorPos, 8, "open")
            task.wait(0.5)
        end
    end
    return false
end

local function CleanAprtParts(entry)
    if not entry then return false end
    local destroyed = 0
    local KEEP = { ["Cooking Pot"] = true, ["Board"] = true, ["backboard"] = true, ["name"] = true }
    local potPart = entry:FindFirstChild("Cooking Pot", true)
    local potPos = potPart and (potPart:IsA("BasePart") and potPart.Position or (potPart:IsA("Model") and potPart:GetPivot().Position)) or nil
    local KEEP_R = 3

    local function isKeep(part)
        local node = part
        repeat
            if KEEP[node.Name] then return true end
            node = node.Parent
        until not node or node == entry
        for _, d in ipairs(part:GetDescendants()) do
            if d:IsA("BasePart") or d:IsA("Model") then
                if KEEP[d.Name] then return true end
            end
        end
        return false
    end

    for _, part in ipairs(entry:GetDescendants()) do
        if not (part:IsA("BasePart") or part:IsA("Model") or part:IsA("Folder")) then continue end
        if isKeep(part) then continue end
        if potPos and part:IsA("BasePart") then
            local dx = part.Position.X - potPos.X
            local dz = part.Position.Z - potPos.Z
            local under = potPos.Y - part.Position.Y
            if dx * dx + dz * dz <= KEEP_R * KEEP_R and under > -1 and under < 6 then continue end
        end
        pcall(function() part:Destroy() end)
        destroyed = destroyed + 1
    end
    return destroyed > 0
end

local function FindApartmentContainer()
    if not _G.OwnedKitchenPos then return nil end
    local bestPot, bestDist = nil, 40
    for _, pot in ipairs(Workspace:GetDescendants()) do
        if pot.Name == "Cooking Pot" then
            local potPos
            if pot:IsA("BasePart") then potPos = pot.Position
            elseif pot:IsA("Model") then potPos = pot:GetPivot().Position end
            if potPos then
                local d = (potPos - _G.OwnedKitchenPos).Magnitude
                if d < bestDist then bestDist = d; bestPot = pot end
            end
        end
    end
    if not bestPot then return nil end
    return bestPot:FindFirstAncestorOfClass("Model")
end

local function SetupApartment()
    -- Cek apakah udah punya apartment
    local ownedData = nil
    for _, data in ipairs(ApartmentData) do
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("TextLabel") then
                local txt = string.upper(obj.Text or "")
                if txt == string.upper(LocalPlayer.Name) then
                    local gui = obj:FindFirstAncestorOfClass("SurfaceGui") or obj:FindFirstAncestorOfClass("BillboardGui")
                    local uiPart = gui and (gui.Adornee or gui.Parent)
                    if uiPart and uiPart:IsA("BasePart") and (uiPart.Position - data.BuyPos).Magnitude <= 8 then
                        ownedData = data
                        break
                    end
                end
            end
        end
        if ownedData then break end
    end

    if ownedData then
        _G.OwnedKitchenPos = ownedData.KitchenPos
        _G.OwnedDoorPos = ownedData.DoorPos
        _G.NoApartmentFull = false
        instantRespawnTP(ownedData.DoorPos)
        task.wait(0.5)
        secureDoor(ownedData.DoorPos)
        task.wait(0.3)
        local aptEntry = FindApartmentContainer()
        if aptEntry then CleanAprtParts(aptEntry) end
        task.wait(0.4)
        return SpawnMotorAfterLockDoor()
    end

    -- Scan VACANT dengan retry
    local targetData = nil
    local MAX_SCAN_ATTEMPTS = 3

    for scanAttempt = 1, MAX_SCAN_ATTEMPTS do
        if getgenv().STOP_ALL then return false end

        for _, data in ipairs(ApartmentData) do
            local isVacant = false
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("TextLabel") and string.find(string.upper(obj.Text or ""), "VACANT") then
                    local gui = obj:FindFirstAncestorOfClass("SurfaceGui") or obj:FindFirstAncestorOfClass("BillboardGui")
                    local uiPart = gui and (gui.Adornee or gui.Parent)
                    if uiPart and uiPart:IsA("BasePart") and (uiPart.Position - data.BuyPos).Magnitude <= 5 then
                        isVacant = true
                        break
                    end
                end
            end
            if isVacant then targetData = data; break end
        end

        if targetData then break end
        task.wait(1)
    end

    if not targetData then
        _G.NoApartmentFull = true
        return false
    end

    -- Retry purchase apartment
    local MAX_PURCHASE_ATTEMPTS = 5
    local purchased = false

    for attempt = 1, MAX_PURCHASE_ATTEMPTS do
        if getgenv().STOP_ALL then return false end

        instantRespawnTP(targetData.BuyPos)
        task.wait(0.8)
        firePromptAt(targetData.BuyPos, 6, "purchase")
        task.wait(1.5)

        local purchasedOK = false
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("TextLabel") then
                local txt = string.upper(obj.Text or "")
                if txt == string.upper(LocalPlayer.Name) then
                    local gui = obj:FindFirstAncestorOfClass("SurfaceGui") or obj:FindFirstAncestorOfClass("BillboardGui")
                    local uiPart = gui and (gui.Adornee or gui.Parent)
                    if uiPart and uiPart:IsA("BasePart") and (uiPart.Position - targetData.BuyPos).Magnitude <= 8 then
                        purchasedOK = true
                        break
                    end
                end
            end
        end

        if purchasedOK then
            purchased = true
            break
        end
        task.wait(1)
    end

    if not purchased then
        _G.NoApartmentFull = true
        return false
    end

    _G.NoApartmentFull = false
    _G.OwnedKitchenPos = targetData.KitchenPos
    _G.OwnedDoorPos = targetData.DoorPos

    blinkTeleport(targetData.DoorPos, false)
    task.wait(0.7)
    secureDoor(targetData.DoorPos)

    task.wait(0.5)
    local aptEntry = FindApartmentContainer()
    if aptEntry then CleanAprtParts(aptEntry) end
    task.wait(0.4)

    return SpawnMotorAfterLockDoor()
end

local function af_FirePrompt(prompt)
    if not prompt or not prompt.Parent then return end
    local orig = RegisterPrompt(prompt)
    pcall(function()
        prompt.MaxActivationDistance = 9999
        prompt.RequiresLineOfSight = false
        prompt.HoldDuration = 0
        task.wait(0.03)
        fireproximityprompt(prompt)
        task.wait(0.06)
    end)
    if orig then
        pcall(function()
            prompt.MaxActivationDistance = orig.MaxActivationDistance
            prompt.RequiresLineOfSight   = orig.RequiresLineOfSight

        end)

local function af_FindPromptByName(keyword, maxDist)
    local charPos = getCharacterPosition()
    if not charPos then return nil end
    keyword = tostring(keyword):lower()
    maxDist = maxDist or 25
    local best, bestDist = nil, maxDist
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and obj.Enabled then
            local parent = obj.Parent
            local pPos
            if parent then
                if parent:IsA("BasePart") then pPos = parent.Position
                elseif parent:IsA("Attachment") then pPos = parent.WorldPosition
                elseif parent:IsA("Model") then pPos = parent:GetPivot().Position
                end
                if not pPos and parent.Parent then
                    if parent.Parent:IsA("BasePart") then pPos = parent.Parent.Position
                    elseif parent.Parent:IsA("Model") then pPos = parent.Parent:GetPivot().Position
                    end
                end
            end
            if pPos then
                local nameChain = ""
                local p = parent
                local depth = 0
                while p and p ~= Workspace and depth < 6 do
                    nameChain = nameChain .. " " .. tostring(p.Name):lower()
                    p = p.Parent
                    depth = depth + 1
                end
                local d = (pPos - charPos).Magnitude
                if d <= bestDist and nameChain:find(keyword, 1, true) then
                    bestDist = d
                    best = obj
                end
            end
        end
    end
    return best
end

local function af_FirePromptByName(keyword, maxDist)
    local prompt = af_FindPromptByName(keyword, maxDist)
    if prompt then af_FirePrompt(prompt); return true end
    return false
end

local function waitForTimer(pot, maxWait, checkInterval)
    checkInterval = checkInterval or 0.3
    local elapsed = 0
    while elapsed < maxWait do
        if getgenv().STOP_ALL then return false end
        local timer = pot:FindFirstChild("Timer")
        if not timer then
            for _, child in pairs(pot:GetChildren()) do
                if child:IsA("TextLabel") and child.Name == "Timer" then timer = child; break end
            end
        end
        if timer and timer:IsA("TextLabel") and (timer.Text or "") == "0" then return true end
        task.wait(checkInterval)
        elapsed = elapsed + checkInterval
    end
    return true
end

local PotCache = {}
local PotCacheKitchen = nil
local PotCacheTime = 0

local function IsPotInUse(pot)
    if not pot or not pot.Parent then return true end
    local timer = pot:FindFirstChild("Timer")
    if not timer then
        for _, child in pairs(pot:GetDescendants()) do
            if child:IsA("TextLabel") and child.Name == "Timer" then timer = child; break end
        end
    end
    if timer and timer:IsA("TextLabel") then
        local txt = (timer.Text or ""):gsub("%s", "")
        if txt == "0" or txt == "" then return false end
        return true
    end
    for _, attrName in ipairs({ "Cooking", "InUse", "Occupied", "IsCooking", "IsUsed" }) do
        local ok, val = pcall(function() return pot:GetAttribute(attrName) end)
        if ok and val ~= nil then
            if val == true then return true end
            if val == false then return false end
        end
    end
    return false
end

local function ScanPotsNear(kitchenPos, radius)
    radius = radius or 15
    local list = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        local isPot = false
        if obj.Name == "Cooking Pot" then isPot = true
        elseif (obj:IsA("BasePart") or obj:IsA("Model")) and obj.Name:lower():find("pot", 1, true) then isPot = true
        end
        if isPot then
            local pos = obj:IsA("BasePart") and obj.Position or (obj:IsA("Model") and obj:GetPivot().Position)
            if pos then
                local d = (pos - kitchenPos).Magnitude
                if d <= radius then
                    local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
                    if not prompt then
                        local att = obj:FindFirstChild("Attachment")
                        if att then prompt = att:FindFirstChildOfClass("ProximityPrompt") end
                    end
                    table.insert(list, { obj = obj, pos = pos, dist = d, prompt = prompt, inUse = IsPotInUse(obj) })
                end
            end
        end
    end
    table.sort(list, function(a, b) return a.dist < b.dist end)
    return list
end

local function RefreshPotCache(kitchenPos, radius)
    PotCache = ScanPotsNear(kitchenPos, radius)
    PotCacheKitchen = kitchenPos
    PotCacheTime = os.clock()
    return PotCache
end

local function GetAvailablePot(kitchenPos, radius)
    radius = radius or 15
    if not PotCacheKitchen or PotCacheKitchen ~= kitchenPos or #PotCache == 0 or (os.clock() - PotCacheTime) > 5 then
        RefreshPotCache(kitchenPos, radius)
    end
    local function pickFree()
        for _, entry in ipairs(PotCache) do
            local pot = entry.obj
            if pot and pot.Parent and not IsPotInUse(pot) then
                local prompt = entry.prompt
                if not prompt or not prompt.Parent then
                    prompt = pot:FindFirstChildWhichIsA("ProximityPrompt", true)
                    if not prompt then
                        local att = pot:FindFirstChild("Attachment")
                        if att then prompt = att:FindFirstChildOfClass("ProximityPrompt") end
                    end
                    entry.prompt = prompt
                end
                if prompt and prompt.Parent then entry.inUse = false; return pot, prompt end
            end
        end
        return nil, nil
    end
    local pot, prompt = pickFree()
    if pot then return pot, prompt end
    RefreshPotCache(kitchenPos, radius)
    pot, prompt = pickFree()
    if pot then return pot, prompt end
    if #PotCache > 0 and PotCache[1].obj and PotCache[1].obj.Parent then
        return PotCache[1].obj, PotCache[1].prompt
    end
    return nil, nil
end

local function MarkPotFree(pot)

local function FirePromptSafe(prompt)
    if not prompt or not prompt.Parent then return false end
    if not prompt.Enabled then return false end
    local orig = RegisterPrompt(prompt)
    pcall(function()
        prompt.HoldDuration          = 0
        prompt.RequiresLineOfSight   = false
        prompt.MaxActivationDistance = 99999
    end)
    local ok1 = pcall(function() fireproximityprompt(prompt) end)
    if not ok1 then
        pcall(function()
            prompt:InputHoldBegin()
            task.wait(0.08)
            prompt:InputHoldEnd()
        end)
    end
    if orig then
        pcall(function()
            prompt.HoldDuration          = orig.HoldDuration
            prompt.RequiresLineOfSight   = orig.RequiresLineOfSight
            prompt.MaxActivationDistance = orig.MaxActivationDistance
        end)
    end
    return true
end

local function _extractTimerNum(lbl)
    if not lbl or not lbl:IsA("TextLabel") then return nil end
    local txt = (lbl.Text or ""):gsub("%s", "")
    if txt == "" then return nil end
    if txt == "0" then return "0" end
    if tonumber(txt) then return txt end
    local num = txt:match("^(%d+)")
    return num
end

local function GetGuiWorldPos(gui)
    if not gui then return nil end
    local function posOf(obj)
        if not obj then return nil end
        if obj:IsA("BasePart") then return obj.Position end
        if obj:IsA("Attachment") then return obj.WorldPosition end
        if obj:IsA("Model") then
            local ok, pv = pcall(function() return obj:GetPivot() end)
            if ok and pv then return pv.Position end
        end
        return nil
    end
    local adPos = posOf(gui.Adornee)
    if adPos then return adPos end
    local p = gui.Parent
    local pPos = posOf(p)
    if pPos then return pPos end
    if p and p.Parent then
        local p2 = posOf(p.Parent)
        if p2 then return p2 end
        if p:IsA("Attachment") and p.Parent:IsA("BasePart") then return p.Parent.Position end
    end
    return nil
end

local function RefreshBillboardMap()
    _billboardMap = {}
    local folder = GetLabPotsFolder()
    if not folder then return end
    local candidates = {}
    for _, d in ipairs(Workspace:GetDescendants()) do
        local isGui = d:IsA("BillboardGui") or d:IsA("SurfaceGui")
        if isGui and d.Enabled then
            local gpos = GetGuiWorldPos(d)
            if gpos then
                local lbl
                for _, l in ipairs(d:GetDescendants()) do
                    if l:IsA("TextLabel") then
                        local n = _extractTimerNum(l)
                        if n then lbl = l; break end
                    end
                end
                if lbl then table.insert(candidates, {pos = gpos, label = lbl}) end
            end
        end
    end
    for _, pot in ipairs(folder:GetChildren()) do
        local pPos = GetPotPosQuick(pot)
        if pPos then
            local bestLbl, bestD = nil, 20
            for _, c in ipairs(candidates) do
                local dd = (c.pos - pPos).Magnitude
                if dd < bestD then bestD = dd; bestLbl = c.label end
            end
            if bestLbl then _billboardMap[pot] = bestLbl end
        end
    end
    _billboardMapT = os.clock()
end

local function GetPotTimerText(pot)
    if not pot or not pot.Parent then return nil end
    local cached = _potTimerCache[pot]
    if cached and cached.Parent then
        local t = _extractTimerNum(cached)
        if t then return t end
    end
    local direct = pot:FindFirstChild("Timer")
    if direct and direct:IsA("TextLabel") then
        _potTimerCache[pot] = direct
        local txt = (direct.Text or ""):gsub("%s", "")
        if txt ~= "" then return txt end
    end
    for _, d in ipairs(pot:GetDescendants()) do
        if d:IsA("TextLabel") and d.Name == "Timer" then
            _potTimerCache[pot] = d
            local txt = (d.Text or ""):gsub("%s", "")
            if txt ~= "" then return txt end
        end
    end
    for _, d in ipairs(pot:GetDescendants()) do
        if (d:IsA("BillboardGui") or d:IsA("SurfaceGui")) and d.Enabled then
            for _, lbl in ipairs(d:GetDescendants()) do
                if lbl:IsA("TextLabel") then
                    local n = _extractTimerNum(lbl)
                    if n then _potTimerCache[pot] = lbl; return n end
                end
            end
        end
    end
    if (os.clock() - _billboardMapT) > _BILLBOARD_TTL then RefreshBillboardMap() end
    local mapped = _billboardMap[pot]
    if mapped and mapped.Parent then
        local n = _extractTimerNum(mapped)
        if n then _potTimerCache[pot] = mapped; return n end
    end
    return nil
end

local function IsPotFreeQuick(pot)
    if not pot or not pot.Parent then return false end
    local txt = GetPotTimerText(pot)
    if txt ~= nil and txt ~= "" and txt ~= "0" then return false end
    for _, attrName in ipairs({ "Cooking", "InUse", "Occupied", "IsCooking", "IsUsed" }) do
        local ok, val = pcall(function() return pot:GetAttribute(attrName) end)
        if ok and val == true then return false end
    end
    return true
end

    local function HasCookNotif()
        local KEYWORDS = {
            "wait for your product",
            "finish cooking in the pot",
            "finish cooking",
            "cooking in the pot",
            "your product",
            "product to finish",
        }
        local function stripRichText(s)
            s = tostring(s or "")
            s = s:gsub("<[^>]->", "")
            return s
        end
        local function matches(s)
            if not s then return false end
            local t = string.lower(stripRichText(s))
            if t == "" then return false end
            for _, kw in ipairs(KEYWORDS) do
                if t:find(kw, 1, true) then return true end
            end
            return false
        end
        local function scan(container)
            if not container then return false end
            local ok, list = pcall(function() return container:GetDescendants() end)
            if not ok or not list then return false end
            for _, d in ipairs(list) do
                local cls = d.ClassName
                if cls == "TextLabel" or cls == "TextButton" or cls == "TextBox" then
                    local ok1, txt = pcall(function() return d.Text end)
                    if ok1 and matches(txt) then return true end
                    local ok2, ctx = pcall(function() return d.ContentText end)
                    if ok2 and matches(ctx) then return true end
                end
            end
            return false
        end
        if scan(LocalPlayer:FindFirstChild("PlayerGui")) then return true end
        if scan(game:GetService("CoreGui")) then return true end
        pcall(function()
            if typeof(gethui) == "function" then
                local hidden = gethui()
                if hidden and scan(hidden) then return true end
            end
        end)
        if scan(Workspace) then return true end
        return false
    end

    local function FindAllPotsInWorkspace()

                       and not nm:find("potato", 1, true)
                       and not nm:find("potion", 1, true)

                local pos = GetPotPosQuick(obj)

        if not success then success = af_FirePromptByName(npcName, 25) end
        task.wait(0.5)
        unequipTools()
        return success
    end

    local MaxRetries = 3
    local retry = 0
    while HasHotChips() and retry < MaxRetries and not getgenv().STOP_ALL do
        retry = retry + 1
        for i, entry in ipairs(HOMELESS_LIST) do
            if not HasHotChips() then break end
            if getgenv().STOP_ALL then break end
            local beforeChips = CountHotChips()
            FeedOneHomelessByCoord(entry)
            task.wait(0.4)
            local afterChips = CountHotChips()
            local consumed = beforeChips - afterChips
            if consumed > 0 then UI.AddStat("Chips Fed", consumed) end
            task.wait(0.3)
        end
        if HasHotChips() then task.wait(1.5) end
    end

    pcall(function()
        local atmGui = PlayerGui:FindFirstChild("ATM")
        if atmGui then atmGui:Destroy() end
    end)
    return true
end

-- RUN CYCLE
local function runCycle()
    ClearLabPots()
    pcall(RefreshBillboardMap)

    if not Step1_ClaimAndWater() or getgenv().STOP_ALL then return false end
    if not Step3_FakeIDAndBank() or getgenv().STOP_ALL then return false end
    if not Step2_LabKentang() or getgenv().STOP_ALL then return false end
    if not Step4_MS_Finish() or getgenv().STOP_ALL then return false end
    if not Step5_GetCardAndSwipe() or getgenv().STOP_ALL then return false end
    if not Step6_HotChipsAndHomeless() or getgenv().STOP_ALL then return false end

    getgenv().CYCLE_COUNT = getgenv().CYCLE_COUNT + 1
    task.wait(1.2)
    return true
end

getgenv().MF_Start = function()
    if getgenv().RUNNING then return end
    getgenv().FarmStartTime = os.clock()
    getgenv().STOP_ALL = false
    getgenv().RUNNING = true
    getgenv().CYCLE_COUNT = 0
    task.spawn(function()
        while not getgenv().STOP_ALL do
            local ok, err = pcall(runCycle)
            if not ok then task.wait(2) end
            if getgenv().STOP_ALL then break end
        end
        getgenv().RUNNING = false
        pcall(stopLockMode)
    end)
end

getgenv().MF_Stop = function()
    getgenv().STOP_ALL = true
    getgenv().FarmStartTime = nil
    pcall(stopLockMode)
    pcall(StopAutoDrive)
    pcall(stopGhostMode)
    pcall(stopGhostModeFully)
    pcall(UnlockMotorPosition)

    _BLINK_NOCLIP_ALL = false

    end)
    -- Reset spawn motor cooldown
    pcall(function()

    end)
end
